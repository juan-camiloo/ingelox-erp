sequenceDiagram
  autonumber
  actor COM as 💼 Asesor Comercial
  actor OP as 🚚 Operador de Tráfico
  participant UI_COT as Cotizaciones UI (/quotations)
  participant UI_INST as Instrucciones UI (/instructions)
  participant UI_TRAF as Tráfico UI (/shipments/new)
  participant SRV as Practiko Dual-Store Service
  participant DB as Supabase PostgreSQL
  participant PORTAL as 🌐 Portal Tracking Clientes

  Note over COM, DB: FASE 1: Creación y Aprobación de la Cotización
  COM->>UI_COT: Ingresa ítems, tarifas (venta/costo), cliente y condiciones
  UI_COT->>SRV: saveQuotation(quotationPayload)
  SRV->>DB: INSERT INTO quotations + quotation_items
  DB-->>SRV: Confirmación (id_cotizacion, consecutivo)
  SRV-->>UI_COT: Cotización guardada en estado 'enviada'
  COM->>UI_COT: Cambia estado a 'aprobada'
  UI_COT->>SRV: updateQuotationStatus(id, 'aprobada')
  SRV->>DB: UPDATE quotations SET estado = 'aprobada'

  Note over COM, OP: FASE 2: Transferencia 1-Clic a Instrucción de Embarque
  COM->>UI_COT: Clic en "Generar Instrucción"
  UI_COT->>UI_INST: Redirecciona a /instructions/new?quotationId=COT-101
  UI_INST->>SRV: getQuotationById(COT-101)
  SRV-->>UI_INST: Datos precargados: Cliente, Asesor, Vía, Tarifas y Notas
  OP->>UI_INST: Completa Incoterm, Ruta (ej. BOG - GDL) y comentarios operativos
  OP->>UI_INST: Clic en "Guardar Instrucción"
  UI_INST->>SRV: saveInstruction(instructionPayload)
  SRV->>DB: INSERT INTO shipping_instructions + concepts
  DB-->>SRV: Instrucción creada (do_number: EXP-26042)

  Note over OP, PORTAL: FASE 3: Apertura en Control de Tráfico & Asignación
  OP->>UI_INST: Clic en "Abrir en Tráfico"
  UI_INST->>UI_TRAF: Redirecciona a /shipments/new?instructionId=EXP-26042
  UI_TRAF->>SRV: getInstructionById(EXP-26042)
  SRV-->>UI_TRAF: Precarga DO, Modo (air/maritime), Origen/Destino, Consignatario
  OP->>UI_TRAF: Selecciona Empresa Portal Tracking y confirma fechas ETD/ETA
  OP->>UI_TRAF: Clic en "Guardar Operación"
  UI_TRAF->>DB: INSERT INTO shipments (do_number, type, origin, destination, etc.)
  DB-->>UI_TRAF: Expediente de Tráfico Creado (id_shipment)
  UI_TRAF->>DB: INSERT INTO company_shipment (company_id, shipment_id)
  UI_TRAF->>DB: INSERT INTO shipment_updates (Apertura de expediente con notas)
  DB-->>PORTAL: Carga inmediatamente visible en el portal de seguimiento del cliente!

-- ═══════════════════════════════════════════════════════════════════
-- INGELOX ERP & TRACKING SYSTEM — SCHEMA DDL POSTGRESQL 15
-- Compatibilidad Completa con Producción y Módulos Comerciales
-- ═══════════════════════════════════════════════════════════════════

-- 1. TABLA: companies (Portal de Tracking en Producción)
CREATE TABLE IF NOT EXISTS public.companies (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. TABLA: profiles (Extensión de Auth Users)
CREATE TABLE IF NOT EXISTS public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email VARCHAR(120) NOT NULL UNIQUE,
  nickname VARCHAR(80),
  is_internal BOOLEAN NOT NULL DEFAULT FALSE,
  company_id UUID REFERENCES public.companies(id) ON DELETE SET NULL,
  status VARCHAR(20) DEFAULT 'active',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. TABLA: shipments (Expedientes de Tráfico DO)
CREATE TABLE IF NOT EXISTS public.shipments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  do_number VARCHAR(50) NOT NULL UNIQUE,
  origin VARCHAR(100) NOT NULL,
  destination VARCHAR(100) NOT NULL,
  shipment_type VARCHAR(20) DEFAULT 'maritime',
  current_status VARCHAR(50) NOT NULL DEFAULT 'expo_start_operation',
  incoterm VARCHAR(20),
  etd DATE,
  eta DATE,
  atd TIMESTAMPTZ,
  ata TIMESTAMPTZ,
  documentary_cutoff TIMESTAMPTZ,
  booking_status VARCHAR(30) DEFAULT 'confirmed',
  inspection_status VARCHAR(30) DEFAULT 'none',
  cargo_type VARCHAR(50) DEFAULT 'general',
  free_days INTEGER,
  carrier VARCHAR(100),
  flight_vessel VARCHAR(100),
  air_waybill VARCHAR(50),
  container_number VARCHAR(50),
  exporter VARCHAR(150),
  consignee VARCHAR(150),
  status VARCHAR(20) DEFAULT 'active',
  created_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  updated_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. TABLA: company_shipment (Asociación N:M Portal Tracking)
CREATE TABLE IF NOT EXISTS public.company_shipment (
  company_id UUID NOT NULL REFERENCES public.companies(id) ON DELETE CASCADE,
  shipment_id UUID NOT NULL REFERENCES public.shipments(id) ON DELETE CASCADE,
  PRIMARY KEY (company_id, shipment_id)
);

-- 5. TABLA: shipment_updates (Historial Inmutable de Novedades)
CREATE TABLE IF NOT EXISTS public.shipment_updates (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  shipment_id UUID NOT NULL REFERENCES public.shipments(id) ON DELETE CASCADE,
  status VARCHAR(50) NOT NULL,
  observation TEXT,
  description TEXT,
  location VARCHAR(150),
  media_url TEXT,
  event_type VARCHAR(50) DEFAULT 'status_update',
  created_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  updated_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. TABLA: documents (Soportes Documentales)
CREATE TABLE IF NOT EXISTS public.documents (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  shipment_id UUID NOT NULL REFERENCES public.shipments(id) ON DELETE CASCADE,
  file_name VARCHAR(255) NOT NULL,
  file_size BIGINT,
  file_path VARCHAR(255),
  storage_path VARCHAR(255),
  created_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. TABLAS COMERCIALES: Practiko ERP
CREATE TABLE IF NOT EXISTS public.third_parties (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tipo_persona VARCHAR(20) NOT NULL CHECK (tipo_persona IN ('juridica', 'natural')),
  tipo_id VARCHAR(20) NOT NULL DEFAULT 'NIT',
  identificacion VARCHAR(30) NOT NULL,
  dv VARCHAR(5),
  razon_social VARCHAR(200) NOT NULL,
  direccion VARCHAR(100),
  telefono VARCHAR(30),
  pais VARCHAR(50) DEFAULT 'Colombia',
  ciudad VARCHAR(80) DEFAULT 'Bogotá D.C.',
  correo_fe VARCHAR(150),
  vendedor VARCHAR(120),
  dias_credito INTEGER DEFAULT 0,
  es_cliente BOOLEAN DEFAULT TRUE,
  es_proveedor BOOLEAN DEFAULT FALSE,
  status VARCHAR(20) DEFAULT 'active',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.billing_concepts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  codigo VARCHAR(50) UNIQUE NOT NULL,
  nombre VARCHAR(150) NOT NULL,
  aplica_iva BOOLEAN DEFAULT FALSE,
  porcentaje_iva NUMERIC DEFAULT 0,
  moneda_default VARCHAR(10) DEFAULT 'USD',
  valor_sugerido NUMERIC DEFAULT 0,
  categoria VARCHAR(50) DEFAULT 'Flete',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.quotations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  consecutivo VARCHAR(50) UNIQUE NOT NULL,
  fecha DATE DEFAULT CURRENT_DATE,
  validez DATE,
  forma_pago VARCHAR(50) DEFAULT 'A 30 DIAS',
  vendedor VARCHAR(120),
  tercero_id UUID REFERENCES public.third_parties(id) ON DELETE SET NULL,
  cliente_nombre VARCHAR(200) NOT NULL,
  subtotal NUMERIC NOT NULL DEFAULT 0,
  total_iva NUMERIC NOT NULL DEFAULT 0,
  total NUMERIC NOT NULL DEFAULT 0,
  moneda VARCHAR(10) DEFAULT 'USD',
  estado VARCHAR(20) DEFAULT 'enviada',
  created_by UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.quotation_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  quotation_id UUID NOT NULL REFERENCES public.quotations(id) ON DELETE CASCADE,
  orden INTEGER NOT NULL DEFAULT 1,
  cantidad NUMERIC NOT NULL DEFAULT 1,
  concepto VARCHAR(150) NOT NULL,
  valor_unitario NUMERIC NOT NULL DEFAULT 0,
  subtotal NUMERIC NOT NULL DEFAULT 0,
  iva NUMERIC NOT NULL DEFAULT 0,
  total NUMERIC NOT NULL DEFAULT 0
);

-- ÍNDICES DE VELOCIDAD
CREATE INDEX IF NOT EXISTS idx_shipments_do ON public.shipments(do_number);
CREATE INDEX IF NOT EXISTS idx_shipments_status ON public.shipments(current_status);
CREATE INDEX IF NOT EXISTS idx_updates_shipment ON public.shipment_updates(shipment_id);
CREATE INDEX IF NOT EXISTS idx_comp_shipment ON public.company_shipment(company_id, shipment_id);
